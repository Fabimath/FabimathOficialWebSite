\newcommand{\caja}[3]{\pgfmathsetmacro\cdx{0.55*#3}\pgfmathsetmacro\cdy{0.38*#3}
  \coordinate (O) at (0,0);\coordinate (X) at (#1,0);\coordinate (Y) at (0,#2);\coordinate (XY) at (#1,#2);
  \coordinate (OD) at (\cdx,\cdy);\coordinate (XD) at ($(X)+(OD)$);\coordinate (YD) at ($(Y)+(OD)$);\coordinate (XYD) at ($(XY)+(OD)$);
  \fill[hielo] (O) rectangle (XY);\fill[narc!30] (Y) -- (YD) -- (XYD) -- (XY) -- cycle;\fill[narc!55] (X) -- (XD) -- (XYD) -- (XY) -- cycle;
  \draw[oculta] (OD) -- (O) (OD) -- (XD) (OD) -- (YD);
  \draw[borde] (O) rectangle (XY) (Y) -- (YD) -- (XYD) -- (XD) -- (X) (XY) -- (XYD);}
