\newcommand{\recto}[3]{\pgfgettransformentries{\rta}{\rtb}{\rtc}{\rtd}{\rtx}{\rty}\pgfmathsetmacro\rsz{0.2/\rta}%
  \coordinate (ra1) at ($(#2)!\rsz cm!(#1)$);\coordinate (ra2) at ($(#2)!\rsz cm!(#3)$);

% \caja{ancho}{alto}{fondo}: paralelepípedo oblicuo con aristas ocultas punteadas.
% Vértices: O X Y XY (cara frontal) y OD XD YD XYD (cara trasera).
\newcommand{\caja}[3]{\pgfmathsetmacro\cdx{0.55*#3}\pgfmathsetmacro\cdy{0.38*#3}
  \coordinate (O) at (0,0);\coordinate (X) at (#1,0);\coordinate (Y) at (0,#2);\coordinate (XY) at (#1,#2);
  \coordinate (OD) at (\cdx,\cdy);\coordinate (XD) at ($(X)+(OD)$);\coordinate (YD) at ($(Y)+(OD)$);\coordinate (XYD) at ($(XY)+(OD)$);
  \fill[hielo] (O) rectangle (XY);\fill[narc!30] (Y) -- (YD) -- (XYD) -- (XY) -- cycle;\fill[narc!55] (X) -- (XD) -- (XYD) -- (XY) -- cycle;
  \draw[oculta] (OD) -- (O) (OD) -- (XD) (OD) -- (YD);
  \draw[borde] (O) rectangle (XY) (Y) -- (YD) -- (XYD) -- (XD) -- (X) (XY) -- (XYD);}
% \cil{radio}{alto}: cilindro vertical, base centrada en (0,0); coordenadas B (centro base) y T (centro tapa).

  \fill[narc!35] (0,#2) ellipse[x radius=#1,y radius=\cry];
  \draw[oculta] (#1,0) arc[start angle=0,end angle=180,x radius=#1,y radius=\cry];
  \draw[borde] (-#1,0) -- (-#1,#2) (#1,0) -- (#1,#2) (0,#2) ellipse[x radius=#1,y radius=\cry]
    (-#1,0) arc[start angle=180,end angle=360,x radius=#1,y radius=\cry];}
